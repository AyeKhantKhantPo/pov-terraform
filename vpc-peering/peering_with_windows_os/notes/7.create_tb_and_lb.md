# Open Windows Firewall for the service port (required for ALB health checks)
# Without this, ALB health checks will timeout even if the service is running
# need to do for each instance
# notes -> 9093 is alb port
New-NetFirewallRule -Name 'FakeService-9093' -DisplayName 'Fake Service Port 9093' -Enabled True -Direction Inbound -Protocol TCP -Action Allow -LocalPort 9093 -Profile Any


New-NetFirewallRule -Name 'FakeService-9092' -DisplayName 'Fake Service Port 9092' -Enabled True -Direction Inbound -Protocol TCP -Action Allow -LocalPort 9092 -Profile Any
