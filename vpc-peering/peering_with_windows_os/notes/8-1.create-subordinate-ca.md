# =============================================================
# Create Subordinate CAs in Tokyo & Osaka signed by Singapore Root CA
# Run all commands from local machine (not inside instances)
# =============================================================

# Singapore Root CA ARN (already active)
ROOT_CA_ARN="arn:aws:acm-pca:ap-southeast-1:439475769687:certificate-authority/9a0fb6eb-83ff-4ef9-b7e0-a65ccd1469e4"

# =============================================================
# PART 1: Tokyo Subordinate CA (for account.hellocloud.io)
# =============================================================

# Step 1: Create the Subordinate CA in Tokyo
TOKYO_SUB_CA_ARN=$(aws acm-pca create-certificate-authority \
  --profile cloud-admin-tokoyo \
  --certificate-authority-configuration '{
    "KeyAlgorithm": "RSA_2048",
    "SigningAlgorithm": "SHA256WITHRSA",
    "Subject": {
      "Organization": "hellocloud",
      "CommonName": "hellocloud.io Tokyo Sub CA"
    }
  }' \
  --certificate-authority-type SUBORDINATE \
  --query 'CertificateAuthorityArn' --output text)

echo "Tokyo Sub CA ARN: $TOKYO_SUB_CA_ARN"

# Step 2: Get CSR from Tokyo Sub CA (wait ~10s after creation)
aws acm-pca get-certificate-authority-csr \
  --profile cloud-admin-tokoyo \
  --certificate-authority-arn "$TOKYO_SUB_CA_ARN" \
  --output text > /tmp/tokyo-sub-ca.csr

# Step 3: Sign the CSR using Singapore Root CA
TOKYO_CERT_ARN=$(aws acm-pca issue-certificate \
  --profile cloud-admin-sgp \
  --certificate-authority-arn "$ROOT_CA_ARN" \
  --csr fileb:///tmp/tokyo-sub-ca.csr \
  --signing-algorithm SHA256WITHRSA \
  --template-arn arn:aws:acm-pca:::template/SubordinateCACertificate_PathLen0/V1 \
  --validity Value=5,Type=YEARS \
  --query 'CertificateArn' --output text)

echo "Tokyo Sub CA Cert ARN: $TOKYO_CERT_ARN"

# Step 4: Wait for cert to be issued, then download it
aws acm-pca wait certificate-issued \
  --profile cloud-admin-sgp \
  --certificate-authority-arn "$ROOT_CA_ARN" \
  --certificate-arn "$TOKYO_CERT_ARN"

aws acm-pca get-certificate \
  --profile cloud-admin-sgp \
  --certificate-authority-arn "$ROOT_CA_ARN" \
  --certificate-arn "$TOKYO_CERT_ARN" \
  --query 'Certificate' --output text > /tmp/tokyo-sub-ca.pem

aws acm-pca get-certificate \
  --profile cloud-admin-sgp \
  --certificate-authority-arn "$ROOT_CA_ARN" \
  --certificate-arn "$TOKYO_CERT_ARN" \
  --query 'CertificateChain' --output text > /tmp/tokyo-sub-ca-chain.pem

# Step 5: Install signed cert back into Tokyo Sub CA — activates it
aws acm-pca import-certificate-authority-certificate \
  --profile cloud-admin-tokoyo \
  --certificate-authority-arn "$TOKYO_SUB_CA_ARN" \
  --certificate fileb:///tmp/tokyo-sub-ca.pem \
  --certificate-chain fileb:///tmp/tokyo-sub-ca-chain.pem

echo "Tokyo Sub CA is now Active"

# Step 6: Request leaf cert directly in Tokyo ACM using Tokyo Sub CA
# Tokyo → Certificate Manager → Request → Request a private certificate
# Select: Tokyo Sub CA, Domain: account.hellocloud.io
# OR via CLI:
aws acm request-certificate \
  --profile cloud-admin-tokoyo \
  --domain-name account.hellocloud.io \
  --certificate-authority-arn "$TOKYO_SUB_CA_ARN" \
  --output json
