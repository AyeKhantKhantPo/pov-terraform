# Step 4: SSH into the instance
ssh profile-instance-02

# --- run the following inside the Windows instance ---
# switch to PowerShell first (cmd does not have Expand-Archive, ls, etc.)
powershell

# Verify nssm is accessible
nssm version

# Step 5: Register fake-service as a Windows service using NSSM
nssm install profile-service "C:\fake-service.exe"

# Step 6: Set environment variables for the service
# NOTE: must be in PowerShell (not cmd). Use $envVars — do NOT use $env (reserved in PowerShell)
# Add UPSTREAM_URIS when the upstream ALB DNS is available
$envVars = "LISTEN_ADDR=0.0.0.0:9091`nNAME=profile-instance-02`nMESSAGE=HelloCloudBank | Retail Banking | Profile Instance 02"
# $envVars += "`nUPSTREAM_URIS=http://<alb-dns>:9092"   # uncomment and fill in ALB DNS when ready

# after lb
$envVars = "LISTEN_ADDR=0.0.0.0:9091`nNAME=profile-instance-02`nMESSAGE=HelloCloudBank | Retail Banking | Profile Instance 02`nUPSTREAM_URIS=http://internal-account-alb-1020001636.ap-northeast-1.elb.amazonaws.com:9092"

# after route 53
$envVars = "LISTEN_ADDR=0.0.0.0:9091`nNAME=profile-instance-01`nMESSAGE=HelloCloudBank | Retail Banking | Profile Instance 01`nUPSTREAM_URIS=http://account.hellocloud.io:9092"

nssm set profile-service AppEnvironmentExtra $envVars

# Verify env vars were saved
nssm get profile-service AppEnvironmentExtra

# Step 7: Start the service
nssm start profile-service
nssm restart profile-service


# Step 8: Verify it is running
nssm status profile-service
netstat -ano | findstr 9091

# Step 9: Test the service
curl http://localhost:9091/
curl http://localhost:9091/health

# Step 10: Open Windows Firewall for the service port (required for ALB health checks)
# Without this, ALB health checks will timeout even if the service is running
New-NetFirewallRule -Name 'FakeService-9091' -DisplayName 'Fake Service Port 9091' -Enabled True -Direction Inbound -Protocol TCP -Action Allow -LocalPort 9091 -Profile Any

# Verify the rule was added
Get-NetFirewallRule -Name 'FakeService-9091' | Select-Object Name, Enabled, Direction, Action
