# NSSM is used to run fake-service as a Windows background service
# Step 1: Download NSSM and save zip + exe to the same folder as fake-service
# NOTE: nssm.cc is down. Download via Chocolatey CDN (contains a nested zip with the actual release):
curl -L -o /home/HninPhyu/HelloCloud/CAP/VPC_Intra_Region_Peering/manual/window_image/nssm.zip "https://community.chocolatey.org/api/v2/package/nssm"
# The Chocolatey package has a nested zip inside — extract it in two steps:
unzip -j /home/HninPhyu/HelloCloud/CAP/VPC_Intra_Region_Peering/manual/window_image/nssm.zip "tools/nssm-2.24-101-g897c7ad.zip" -d /tmp/
unzip -j /tmp/nssm-2.24-101-g897c7ad.zip "nssm-2.24-101-g897c7ad/win64/nssm.exe" -d /home/HninPhyu/HelloCloud/CAP/VPC_Intra_Region_Peering/manual/window_image/
chmod +x nssm.exe


# Step 2: Download fake-service Windows binary on local machine
curl -LO https://github.com/nicholasjackson/fake-service/releases/download/v0.26.2/fake_service_windows_amd64.zip
unzip fake_service_windows_amd64.zip
chmod +x fake-service.exe

# Step 3: Copy both binaries directly to the Windows instance (uses ~/.ssh/config)
scp /home/HninPhyu/HelloCloud/CAP/VPC_Intra_Region_Peering/manual/window_image/nssm.exe jump-host:"C:/Windows/System32/nssm.exe"
scp /home/HninPhyu/HelloCloud/CAP/VPC_Intra_Region_Peering/manual/window_image/fake-service.exe jump-host:"C:/fake-service.exe"
