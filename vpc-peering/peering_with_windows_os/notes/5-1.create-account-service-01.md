# Step 4: SSH into the instance
ssh account-instance-01

# --- run the following inside the Windows instance ---
# switch to PowerShell first (cmd does not have Expand-Archive, ls, etc.)
powershell

# Verify nssm is accessible
nssm version

# Step 5: Register fake-service as a Windows service using NSSM
nssm install account-service "C:\fake-service.exe"

# Step 6: Set environment variables for the service
# NOTE: must be in PowerShell (not cmd). Use $envVars — do NOT use $env (reserved in PowerShell)
# Add UPSTREAM_URIS when the upstream ALB DNS is available
$envVars = "LISTEN_ADDR=0.0.0.0:9092`nNAME=account-instance-01`nMESSAGE=HelloCloudBank | Retail Banking | Account Instance 01"

# after upstream is ready
$envVars = "LISTEN_ADDR=0.0.0.0:9092`nNAME=account-instance-01`nMESSAGE=HelloCloudBank | Retail Banking | Account Instance 01`nUPSTREAM_URIS=http://internal-statement-alb-334019484.ap-northeast-3.elb.amazonaws.com:9093"

# after route 53
$envVars = "LISTEN_ADDR=0.0.0.0:9092`nNAME=account-instance-01`nMESSAGE=HelloCloudBank | Retail Banking | Account Instance 01`nUPSTREAM_URIS=http://statement.hellocloud.io:9093"

nssm set account-service AppEnvironmentExtra $envVars

# Verify env vars were saved
nssm get account-service AppEnvironmentExtra

# Step 7: Start the service
nssm start account-service
nssm restart account-service

# Step 8: Verify it is running
nssm status account-service
netstat -ano | findstr 9092

# Step 9: Test the service
curl http://localhost:9092/
curl http://localhost:9092/health

# Step 10: Open Windows Firewall for the service port (required for ALB health checks)
# Without this, ALB health checks will timeout even if the service is running
New-NetFirewallRule -Name 'FakeService-9092' -DisplayName 'Fake Service Port 9092' -Enabled True -Direction Inbound -Protocol TCP -Action Allow -LocalPort 9092 -Profile Any


# Verify the rule was added
Get-NetFirewallRule -Name 'FakeService-9092' | Select-Object Name, Enabled, Direction, Action
