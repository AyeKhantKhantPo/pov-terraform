# =============================================================
# Route 53 Private Hosted Zone + ACM Private CA + HTTPS Setup
# =============================================================

# -------------------------------------------------------------
# PART 1: Create Private Hosted Zone (Route 53 Console)
# -------------------------------------------------------------
# 1. Route 53 → Hosted zones → Create hosted zone
# 2. Domain name: hellocloud.io
# 3. Type: Private hosted zone
# 4. VPC: associate all 3 VPCs (Singapore, Tokyo, Osaka)
# 5. Create A records (alias) for each service:
#    - profile.hellocloud.io   → profile-alb   (Singapore)
#    - account.hellocloud.io   → account-alb   (Tokyo)
#    - statement.hellocloud.io → statement-alb (Osaka)
# NOTE: if DNS fails after creation, check VPC association —
#       the hosted zone must be associated with the VPC where the instance lives

# -------------------------------------------------------------
# PART 2: Create Private Root CA (Singapore Console — one time)
# -------------------------------------------------------------
# 1. Singapore → Certificate Manager → AWS Private CA → Create a CA
# 2. CA type: Root
# 3. Subject: O=hellocloud, CN=hellocloud.io
# 4. Key algorithm: RSA 2048
# 5. After created → Actions → Install CA certificate → Activate
# 6. Copy the CA ARN

# Request leaf certs from Singapore ACM (one per service domain):
# Singapore → Certificate Manager → Request → Request a private certificate
#   - account.hellocloud.io   (will be imported into Tokyo)
#   - statement.hellocloud.io (will be imported into Osaka)
#   - profile.hellocloud.io   (stays in Singapore, used by profile-alb)

# -------------------------------------------------------------
# PART 3: Export cert from Singapore + Import to target region
# -------------------------------------------------------------
# AWS forces passphrase on export — decrypt immediately after.
# Do this for account (Tokyo) and statement (Osaka) certs.

# Step 1: Export from Singapore ACM
# NOTE: --passphrase must be base64-encoded and at least 4 chars
# Set your passphrase here (plain text — it gets base64-encoded below)
PASSPHRASE="hellocloud123"
PASSPHRASE_B64=$(echo -n "$PASSPHRASE" | base64)

# --- account.hellocloud.io → Tokyo ---
aws acm export-certificate --profile cloud-admin-sgp \
  --certificate-arn arn:aws:acm:ap-southeast-1:439475769687:certificate/41375521-f07d-4e43-9aa8-a2cf500afa34 \
  --passphrase "$PASSPHRASE_B64" \
  --output json > /tmp/account-cert-raw.json

# --- statement.hellocloud.io → Osaka ---
aws acm export-certificate --profile cloud-admin-sgp \
  --certificate-arn arn:aws:acm:ap-southeast-1:439475769687:certificate/23caf9e5-f107-4c53-9da4-2815ee08368b \
  --passphrase "$PASSPHRASE_B64" \
  --output json > /tmp/statement-cert-raw.json

# Step 2: Parse JSON into separate PEM files
for PREFIX in account statement; do
  python3 -c "
import json
d = json.load(open('/tmp/${PREFIX}-cert-raw.json'))
open('/tmp/${PREFIX}-cert.pem','w').write(d['Certificate'])
open('/tmp/${PREFIX}-key-enc.pem','w').write(d['PrivateKey'])
open('/tmp/${PREFIX}-chain.pem','w').write(d['CertificateChain'])
"
done

# Step 3: Decrypt private keys (ACM import requires unencrypted key)
openssl pkey -in /tmp/account-key-enc.pem   -out /tmp/account-key.pem   -passin pass:"$PASSPHRASE"
openssl pkey -in /tmp/statement-key-enc.pem -out /tmp/statement-key.pem -passin pass:"$PASSPHRASE"

# Step 4: Import into target region ACM
aws acm import-certificate --profile cloud-admin-tokoyo \
  --certificate  fileb:///tmp/account-cert.pem \
  --private-key  fileb:///tmp/account-key.pem \
  --certificate-chain fileb:///tmp/account-chain.pem \
  --output json

aws acm import-certificate --profile cloud-admin-osaka \
  --certificate  fileb:///tmp/statement-cert.pem \
  --private-key  fileb:///tmp/statement-key.pem \
  --certificate-chain fileb:///tmp/statement-chain.pem \
  --output json

# -------------------------------------------------------------
# PART 4: Attach certs to ALB listeners (Console)
# -------------------------------------------------------------
# For each ALB: EC2 → Load Balancers → select ALB → Listeners tab
#   → Add listener → HTTPS:443 → Forward to target group
#   → Default SSL/TLS cert → From ACM → select the cert
#   → Add
#
# Optionally add HTTP:80 → Redirect to HTTPS:443
#
# profile-alb  (Singapore) → profile.hellocloud.io  cert (already in Singapore ACM, In use)
# account-alb  (Tokyo)     → account.hellocloud.io  cert (imported in Part 3)
# statement-alb(Osaka)     → statement.hellocloud.io cert (imported in Part 3)
