
# NOTE: C:\Temp\ does not exist by default on Windows — create it first via ssh, or use C:\Users\Administrator\
# Option A: create the folder first (run once per instance)
#   ssh profile-instance-01 "mkdir C:\Temp"
# Option B: copy directly to the user home dir (no mkdir needed) — preferred

# echo "Copying to Customer Profile instance 01 ..."
scp certfiles/rootcert.pem profile-instance-01:"C:/Users/Administrator/root-ca.cer"
ssh profile-instance-01

# echo "Copying to Customer Profile instance 02 ..."
scp certfiles/rootcert.pem profile-instance-02:"C:/Users/Administrator/root-ca.cer"
ssh profile-instance-02

# echo "Copying to Account instance 01 ..."
scp certfiles/rootcert.pem account-instance-01:"C:/Users/Administrator/root-ca.cer"
ssh account-instance-01

# echo "Copying to Account instance 02 ..."
scp certfiles/rootcert.pem account-instance-02:"C:/Users/Administrator/root-ca.cer"
ssh account-instance-02

# open powershell
Import-Certificate -FilePath "C:\Users\Administrator\root-ca.cer" -CertStoreLocation "Cert:\LocalMachine\Root"

# verify the cert is installed
Get-ChildItem Cert:\LocalMachine\Root | Where-Object {$_.Subject -like "*HelloCloud*"}

Get-ChildItem Cert:\LocalMachine\Root

# Restart-Service <YourServiceName>
nssm restart profile-service
nssm restart account-service
nssm restart statement-service
